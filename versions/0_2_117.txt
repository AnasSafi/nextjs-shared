NEW FILE CONTENT (copy/paste):

"use client";

import {useTranslation} from "@/app/i18n/client_translation";
import {createModalHook as createModalHookShared} from "@nextjs-shared/shared";
import type {AllowedUserType} from "@nextjs-shared/shared";
import type {FormPropsModel, FormRefModel} from "@nextjs-shared/shared";

export function createModalHook<TIn, TOut = TIn>(config: {
    modalId: string;
    dialogTitleKey: string;
    FormComponent: React.ForwardRefExoticComponent<
        React.PropsWithoutRef<FormPropsModel<TIn, TOut>> &
        React.RefAttributes<FormRefModel>
    >;
    allowed?: AllowedUserType;
    acceptButtonText?: string;
    withMinHeight?: boolean;
    dialogMaxWidthClass?: string;
    readOnly?: boolean;
    fullHeightScreen?: boolean;
}) {
    const {t} = useTranslation();
    return createModalHookShared<TIn, TOut>({...config, t});
}
