import type {NextRequest} from "next/server";
import {createApiUtils} from "@nextjs-shared/shared";
import type {UserModel, PermissionModel} from "@nextjs-shared/shared";
import {getTranslations} from "@/app/i18n/server_translation";
import JWTUtils from "@/custom_utils/jwtUtils";
import {ServerResponse} from "@/extensions/ServerResponse";
import type {RouteConfigModel} from "@/models/core/routeConfigModes";
import SystemQueries from "@/db/queries/SystemQueries";
import {setAppLocale} from "@/custom_utils/appLocale";
import {getUserFromRequest} from "@/custom_services/user/userServices"; (or import {getUserFromRequest} from "@/services/user/UserServices";)

async function isAuthenticated(request: NextRequest): Promise<boolean> {
    const token: string | null = request.headers.get("Authorization");
    return token ? JWTUtils.isValid(token) : false;
}

async function isSuperAdmin(user: UserModel | null): Promise<boolean> {
    if (user?.id) return user.properties?.isSuperAdmin ?? false;
    return false;
}

async function isUnderMaintenance(): Promise<boolean> {
    const systemQueries = new SystemQueries();
    return systemQueries.isInMaintenanceMode();
}

async function getUserPermissions(user: UserModel | null): Promise<string[]> {
    return user?.permissions.map((p: PermissionModel) => p.permissionKey) ?? [];
}

const api = createApiUtils({
    ServerResponse,
    isServerResponse: (v: unknown): v is ServerResponse => v instanceof ServerResponse,

    getTranslations: async () => {
        const {t} = await getTranslations();
        return {t};
    },
    setAppLocale,

    getUserFromRequest,
    isAuthenticated,
    isSuperAdmin,
    isUnderMaintenance,
    getUserPermissions,
});

export const withErrorHandling = api.withErrorHandling;
export const withAll = <J = unknown>(
    routeFunc: Parameters<typeof api.withAll<J>>[0],
    config: RouteConfigModel & {bodyType?: "json" | "form" | "text" | undefined},
) => api.withAll<J>(routeFunc, config);

export const HttpError = api.HttpError;
export const BadRequestError = api.BadRequestError;
export const assertDefined = api.assertDefined;
