"use client";

import React, {forwardRef, type JSX} from "react";
import type {FieldValues} from "react-hook-form";

import {ComboBoxCore} from "@nextjs-shared/shared";
import type {ComboBoxCoreProps} from "@nextjs-shared/shared";

import {useTranslation} from "@/app/i18n/client_translation";
import {useComboCore} from "@/custom_utils/Form/Helpers/useComboCore";
import {useCreateOptionModal} from "@/custom_hooks/common/useCreateOptionModal";
import {useEditOptionModal} from "@/custom_hooks/common/useEditOptionModal";
import {fetchUtil} from "@/custom_utils/fetchUtils/fetchUtil.client";

function ComboBoxInner<TFormValues extends FieldValues = FieldValues>(
    props: ComboBoxCoreProps<TFormValues>,
    ref: React.ForwardedRef<HTMLInputElement>,
): JSX.Element {
    const {t, currentLng} = useTranslation();
    const {openModal: openCreateOptionModal} = useCreateOptionModal();
    const {openModal: openEditOptionModal} = useEditOptionModal();

    return (
        <ComboBoxCore
            {...props}
            ref={ref}
            t={t}
            currentLng={currentLng}
            useComboCore={useComboCore}
            openCreateOptionModal={openCreateOptionModal}
            openEditOptionModal={openEditOptionModal}
            fetchUtil={fetchUtil}
        />
    );
}

export const ComboBox = forwardRef(ComboBoxInner) as <
    TFormValues extends FieldValues = FieldValues
>(
    props: ComboBoxCoreProps<TFormValues> & {
        ref?: React.ForwardedRef<HTMLInputElement>;
    },
) => JSX.Element;

export default ComboBox;
