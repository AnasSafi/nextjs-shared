"use client";

import React, {ReactElement} from "react";
import {TableFilters} from "@nextjs-shared/shared";
import type {PaginationRequestModel, FilterModel} from "@nextjs-shared/shared";
import {useTranslation} from "@/app/i18n/client_translation";
import CMultiLangInput from "@/custom_components/Form/CMultiLangInput";

export type OpenFilterModalFn = (args: { filters: FilterModel[] }) =>
    Promise<FilterModel[] | null | undefined>;

export type SubmitFiltersFn = (filters: FilterModel[]) => void;

export default function TableFiltersWrapper(
    {
        openFilterModalAction,
        submitFiltersAction,
        pagination,
        generalSearchEnabled = true,
    }: {
        openFilterModalAction?: OpenFilterModalFn;
        submitFiltersAction: SubmitFiltersFn;
        pagination: PaginationRequestModel;
        generalSearchEnabled?: boolean;
    }
): ReactElement {
    const {t, currentLng} = useTranslation();

    return (
        <TableFilters
            openFilterModalAction={openFilterModalAction}
            submitFiltersAction={submitFiltersAction}
            pagination={pagination}
            generalSearchEnabled={generalSearchEnabled}
            currentLng={currentLng}
            translations={{
                generalSearch: t("generalSearch"),
                advancedFilter: t("advancedFilter"),
                resetFilters: t("resetFilters"),
            }}
            SearchInput={CMultiLangInput}
        />
    );
}
