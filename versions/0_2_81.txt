import type {JSX} from "react";
import Link from "next/link";
import Image from "next/image";
import type {Column} from "@nextjs-shared/shared";
import {
    getTableCellContent,
    type TableCellContentRenderers,
} from "@nextjs-shared/shared";

const renderers: TableCellContentRenderers = {
    Link: ({href, children, className, target, download}) => (
        <Link
            href={href}
            target={target}
            download={download}
            className={className}
        >
            {children}
        </Link>
    ),
    Image: ({src, alt, className, width, height}) => (
        <Image
            src={src}
            alt={alt}
            className={className}
            width={width || 100}
            height={height || 100}
        />
    ),
};

export function TableCellContent<T>(
    {
        col,
        row,
        currentLng,
    }: {
        col: Column<T>;
        row: T;
        currentLng: string;
    }): JSX.Element {
    return getTableCellContent({col, row, currentLng, renderers});
}
