"use client";

import React from "react";
import DOMPurify from "isomorphic-dompurify";
import {
    ATTR_BLOCK_RE_FRAGMENT,
    createMarkdownTextView,
    getPreferredIso3LangFromDom,
    getStringFromMultiLangValueByLang,
    isLanguageRTL,
    parseAttrBlock,
    parseMLAttr,
    resolveMediaSrc,
} from "@nextjs-shared/shared";
import {fallbackLng, languages} from "@/config/localizationConfig";

const sanitizeHtml = (html: string): string => {
    return DOMPurify.sanitize(html, {
        ALLOW_DATA_ATTR: true,
        ADD_TAGS: ["figure", "figcaption"],
        ADD_ATTR: ["referrerpolicy"],
    });
};

const MarkdownTextView = createMarkdownTextView({
    sanitizeHtml,
    fallbackLng,
    languages,

    ATTR_BLOCK_RE_FRAGMENT,
    parseAttrBlock,
    parseMLAttr,

    getPreferredIso3LangFromDom,
    getStringFromMultiLangValueByLang,
    isLanguageRTL,

    resolveMediaSrc,
});

export default MarkdownTextView;
