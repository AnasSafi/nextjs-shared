1) Create file:
src/custom_components/Modal/ModalTemplate.tsx

NEW FILE CONTENT (copy/paste):

"use client";

import type {FC, ReactNode} from "react";
import React from "react";
import {ModalTemplateCore} from "@nextjs-shared/shared";
import type {ModalTemplateProps} from "@nextjs-shared/shared";
import {useTranslation} from "@/app/i18n/client_translation";

type Props = Omit<ModalTemplateProps, "t"> & { children: ReactNode };

export const ModalTemplate: FC<Props> = ({id, children}) => {
    const {t} = useTranslation();
    return <ModalTemplateCore id={id} t={t} >{children}</ModalTemplateCore>;
};

export default ModalTemplate;
