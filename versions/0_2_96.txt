import React, {forwardRef, type JSX} from "react";

import {useTranslation} from "@/app/i18n/client_translation";
import {fetchUtil} from "@/custom_utils/fetchUtils/fetchUtil.client";
import {buttonClassNames, themeColor} from "@/config/style";
import {TableApiMethod} from "@/config/tableConfig";
import TableFilters from "@/custom_components/Tables/TableFilters";

import {
    TablePagesCore,
    type RowMeta,
    type TableProps,
    type TablePagesHandle,
} from "@nextjs-shared/shared";

function TablePagesInnerRef<R extends RowMeta<string | number, R>>(
    props: TableProps<R>,
    ref: React.Ref<TablePagesHandle>,
): JSX.Element {
    const {t, currentLng} = useTranslation();

    return (
        <TablePagesCore
            {...props}
            ref={ref}
            t={t}
            currentLng={currentLng}
            fetchUtil={fetchUtil}
            themeColor={themeColor}
            buttonClassNames={buttonClassNames}
            tableApiMethod={TableApiMethod}
            TableFiltersComponent={TableFilters}
        />
    );
}

export const TablePages = forwardRef(TablePagesInnerRef) as <
    R extends RowMeta<string | number, R>,
>(
    p: TableProps<R> & { ref?: React.Ref<TablePagesHandle> },
) => JSX.Element;
